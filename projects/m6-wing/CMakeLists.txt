cmake_minimum_required(VERSION 3.20)
project(m6_combined_equivalent_candidate LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(M6_CELL_JACOBIAN_CACHE "Use the scoped per-cell derivative cache" ON)
option(M6_SKIP_UNUSED_NORMAL_LSQ "Skip overwritten scalar LSQ work for normal SST without frozen limiters" OFF)
option(M6_BUILD_DEAD_NORMAL_LSQ_PROBE "Build the independent small-mesh gradient equivalence probe" OFF)
option(M6_CELL_JACOBIAN_THERMO_CACHE "Reuse immutable cell rho/cp in the scoped approximate Jacobian cache" OFF)
option(M6_DEFER_POSTFLOW_K_RECONSTRUCTION "Skip only the overwritten post-flow normal-k reconstruction" OFF)
option(M6_BUILD_POSTFLOW_K_PROBE "Build the direct caller-stage live-array equivalence probe" OFF)
option(M6_SKIP_DISCARDED_POSTFLOW_ASSEMBLY "Skip discarded normal-SST post-flow assembly, retaining its domain checks" OFF)
option(M6_BUILD_SKIP_POSTFLOW_ASSEMBLY_PROBE "Build the real small-mesh post-flow equivalence probe" OFF)
option(M6_BUILD_SOLVER "Build MPI solver and existing core checks" ON)
if(M6_CELL_JACOBIAN_THERMO_CACHE AND NOT M6_CELL_JACOBIAN_CACHE)
  message(FATAL_ERROR "Cell thermodynamic derivative cache requires the existing cell Jacobian cache")
endif()
enable_testing()
if(MSVC)
  set(CMAKE_CL_SHOWINCLUDES_PREFIX "Note: including file:")
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  set(CMAKE_CXX_COMPILER_LAUNCHER "${Python3_EXECUTABLE};${CMAKE_CURRENT_SOURCE_DIR}/tools/msvc_compile.py")
endif()
add_executable(m6_cell_jacobian_cache_test tools/m6_cell_jacobian_cache_test.cpp)
add_test(NAME m6_cell_jacobian_cache_bits COMMAND m6_cell_jacobian_cache_test)
if(M6_BUILD_SOLVER)
  find_package(MPI REQUIRED COMPONENTS CXX)
  add_executable(m6 src/m6.cpp)
  target_link_libraries(m6 PRIVATE MPI::MPI_CXX)
  if(M6_CELL_JACOBIAN_CACHE)
    target_compile_definitions(m6 PRIVATE M6_CELL_JACOBIAN_CACHE=1)
  endif()
  if(M6_SKIP_UNUSED_NORMAL_LSQ)
    target_compile_definitions(m6 PRIVATE M6_SKIP_UNUSED_NORMAL_LSQ=1)
  endif()
  if(M6_DEFER_POSTFLOW_K_RECONSTRUCTION)
    target_compile_definitions(m6 PRIVATE M6_DEFER_POSTFLOW_K_RECONSTRUCTION=1)
  endif()
  if(M6_BUILD_POSTFLOW_K_PROBE)
    add_executable(m6_postflow_k_pair_probe tools/m6_postflow_k_pair_probe.cpp)
    target_link_libraries(m6_postflow_k_pair_probe PRIVATE MPI::MPI_CXX)
    if(M6_CELL_JACOBIAN_CACHE)
      target_compile_definitions(m6_postflow_k_pair_probe PRIVATE M6_CELL_JACOBIAN_CACHE=1)
    endif()
    if(M6_SKIP_UNUSED_NORMAL_LSQ)
      target_compile_definitions(m6_postflow_k_pair_probe PRIVATE M6_SKIP_UNUSED_NORMAL_LSQ=1)
    endif()
    if(M6_CELL_JACOBIAN_THERMO_CACHE)
      target_compile_definitions(m6_postflow_k_pair_probe PRIVATE M6_CELL_JACOBIAN_THERMO_CACHE=1)
    endif()
    add_test(NAME m6_postflow_k_direct_bits
      COMMAND m6_postflow_k_pair_probe --synthetic "${CMAKE_CURRENT_BINARY_DIR}/postflow-k-direct")
  endif()
  if(M6_SKIP_DISCARDED_POSTFLOW_ASSEMBLY)
    target_compile_definitions(m6 PRIVATE M6_SKIP_DISCARDED_POSTFLOW_ASSEMBLY=1)
  endif()
  add_executable(cylinder src/cylinder.cpp)
  add_executable(m6_wall_slip_probe tools/m6_wall_slip_probe.cpp)
  add_executable(m6_wall_thermal_probe tools/m6_wall_thermal_probe.cpp)
  add_test(NAME numerical_core COMMAND cylinder --self-test)
  add_test(NAME m6_core COMMAND m6 --self-test)
  add_test(NAME m6_wall_slip_core COMMAND m6_wall_slip_probe --self-test)
  add_test(NAME m6_wall_thermal_core COMMAND m6_wall_thermal_probe --self-test)
  if(M6_BUILD_DEAD_NORMAL_LSQ_PROBE)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    add_executable(m6_dead_normal_lsq_probe_off tools/m6_dead_normal_lsq_probe.cpp)
    add_executable(m6_dead_normal_lsq_probe_on tools/m6_dead_normal_lsq_probe.cpp)
    add_executable(m6_dead_normal_lsq_pair_probe tools/m6_dead_normal_lsq_pair_probe.cpp)
    target_link_libraries(m6_dead_normal_lsq_probe_off PRIVATE MPI::MPI_CXX)
    target_link_libraries(m6_dead_normal_lsq_probe_on PRIVATE MPI::MPI_CXX)
    target_link_libraries(m6_dead_normal_lsq_pair_probe PRIVATE MPI::MPI_CXX)
    target_compile_definitions(m6_dead_normal_lsq_probe_on PRIVATE M6_SKIP_UNUSED_NORMAL_LSQ=1)
    add_test(NAME m6_dead_normal_lsq_bits
      COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tools/check_m6_dead_normal_lsq_bits.py"
        "$<TARGET_FILE:m6_dead_normal_lsq_probe_off>" "$<TARGET_FILE:m6_dead_normal_lsq_probe_on>"
        "${CMAKE_CURRENT_BINARY_DIR}/dead-normal-lsq-bits")
    add_test(NAME m6_dead_normal_lsq_direct
      COMMAND m6_dead_normal_lsq_pair_probe --synthetic "${CMAKE_CURRENT_BINARY_DIR}/dead-normal-lsq-direct")
  endif()
  if(M6_BUILD_SKIP_POSTFLOW_ASSEMBLY_PROBE)
    add_executable(m6_skip_postflow_assembly_pair_probe tools/m6_skip_postflow_assembly_pair_probe.cpp)
    target_link_libraries(m6_skip_postflow_assembly_pair_probe PRIVATE MPI::MPI_CXX)
    if(M6_CELL_JACOBIAN_CACHE)
      target_compile_definitions(m6_skip_postflow_assembly_pair_probe PRIVATE M6_CELL_JACOBIAN_CACHE=1)
    endif()
    # Both Solver namespaces use these selected existing optimizations. The
    # probe itself controls only skip OFF/ON; its legacy LSQ reference remains
    # OFF so the original all-output fallback checks retain their meaning.
    if(M6_CELL_JACOBIAN_THERMO_CACHE)
      target_compile_definitions(m6_skip_postflow_assembly_pair_probe PRIVATE M6_CELL_JACOBIAN_THERMO_CACHE=1)
    endif()
    if(M6_DEFER_POSTFLOW_K_RECONSTRUCTION)
      target_compile_definitions(m6_skip_postflow_assembly_pair_probe PRIVATE M6_DEFER_POSTFLOW_K_RECONSTRUCTION=1)
    endif()
    add_test(NAME m6_skip_postflow_assembly_live_bits
      COMMAND m6_skip_postflow_assembly_pair_probe "${CMAKE_CURRENT_BINARY_DIR}/skip-postflow-assembly-probe")
  endif()
endif()
get_property(all_targets DIRECTORY PROPERTY BUILDSYSTEM_TARGETS)
foreach(target IN LISTS all_targets)
  if(M6_CELL_JACOBIAN_THERMO_CACHE AND (target STREQUAL "m6" OR target STREQUAL "m6_cell_jacobian_cache_test"))
    target_compile_definitions(${target} PRIVATE M6_CELL_JACOBIAN_THERMO_CACHE=1)
  endif()
  if(MSVC)
    target_compile_options(${target} PRIVATE /O2 /Zi /fp:precise /W4)
    target_link_options(${target} PRIVATE /DEBUG)
  else()
    target_compile_options(${target} PRIVATE -O3 -g -Wall -Wextra -ffp-contract=off)
  endif()
endforeach()

cmake_minimum_required(VERSION 3.20)
project(cylinder_flow LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_INTERPROCEDURAL_OPTIMIZATION OFF)
enable_testing()
if(MSVC)
  set(CMAKE_CL_SHOWINCLUDES_PREFIX "Note: including file:")
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  set(CMAKE_CXX_COMPILER_LAUNCHER "${Python3_EXECUTABLE};${CMAKE_CURRENT_SOURCE_DIR}/tools/msvc_compile.py")
endif()
add_executable(cylinder src/cylinder.cpp)
add_executable(cylinder_anderson_qr_probe tools/cylinder_anderson_qr_probe.cpp)
foreach(target cylinder cylinder_anderson_qr_probe)
  target_compile_definitions(${target} PRIVATE CFD_CYLINDER_INCREMENTAL_ANDERSON_QR=1 CFD_CYLINDER_SKIP_UNUSED_RADIUS_WORK=1)
  if(MSVC)
    target_compile_options(${target} PRIVATE /O2 /fp:precise /W4)
  else()
    target_compile_options(${target} PRIVATE -O3 -Wall -Wextra -ffp-contract=off)
  endif()
endforeach()
add_test(NAME numerical_core COMMAND cylinder --self-test)
add_test(NAME incremental_anderson_QR_core COMMAND cylinder_anderson_qr_probe)
